function doGet(e) {
  const params = (e && e.parameter) ? e.parameter : {};
  if (!params.action) {
    return HtmlService.createHtmlOutputFromFile('index')
      .setTitle('康橋 × 舊鞋救命｜2026 服務學習')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
      .addMetaTag('viewport', 'width=device-width, initial-scale=1.0');
  }

  let result = { ok: false, error: '未知的請求動作' };
  try {
    if (params.action === 'ping') {
      result = handlePing();
    } else if (params.action === 'syncPull') {
      result = handleSyncPull();
    }
  } catch (err) {
    result = { ok: false, error: err.message || String(err) };
  }

  const jsonStr = JSON.stringify(result);
  if (params.callback) {
    return ContentService.createTextOutput(params.callback + '(' + jsonStr + ');')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(jsonStr)
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  let result = { ok: false, error: '無效的請求內容' };
  try {
    const raw = (e && e.postData && e.postData.contents) ? e.postData.contents : '{}';
    const payload = JSON.parse(raw);
    const action = payload.action || '';

    if (action === 'ping') {
      result = handlePing();
    } else if (action === 'sendOtp') {
      result = handleSendOtp(payload);
    } else if (action === 'verifyOtp') {
      result = handleVerifyOtp(payload);
    } else if (action === 'sendNotice') {
      result = handleSendNotice(payload);
    } else if (action === 'syncPull') {
      result = handleSyncPull();
    } else if (action === 'syncPush') {
      result = handleSyncPush(payload);
    } else {
      result = { ok: false, error: '不支援的操作類型：' + action };
    }
  } catch (err) {
    result = { ok: false, error: err.message || String(err) };
  }

  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function normalizeStudentEmail(studentId, providedEmail) {
  const cleanEmail = String(providedEmail || '').trim().toLowerCase();
  if (cleanEmail && cleanEmail.indexOf('@') > 0) {
    return cleanEmail;
  }
  const cleanId = String(studentId || '').trim().toLowerCase();
  if (!cleanId) return '';
  if (cleanId === 'chunyichou' || cleanId === 'yiyunchen') {
    return cleanId + '@kcislk.ntpc.edu.tw';
  }
  return cleanId + '@stu.kcislk.ntpc.edu.tw';
}

function maskEmailAddress(email) {
  const parts = String(email || '').split('@');
  if (parts.length !== 2) return email;
  const local = parts[0];
  if (local.length <= 3) return local.charAt(0) + '***@' + parts[1];
  return local.slice(0, 3) + '***' + local.slice(-1) + '@' + parts[1];
}

/**
 * 2. 伺服器端驗證學生輸入的 6 位數驗證碼
 */
function handleVerifyOtp(payload) {
  const studentId = String(payload.studentId || '').trim();
  const recipientEmail = normalizeStudentEmail(studentId, payload.email);
  const inputCode = String(payload.code || '').trim();

  if (!recipientEmail || !inputCode) {
    return { ok: false, error: '請輸入完整的 6 位數驗證碼' };
  }

  const cache = CacheService.getScriptCache();
  const key = 'OTP_DATA_' + recipientEmail;
  const raw = cache.get(key);
  if (!raw) {
    return { ok: false, error: '驗證碼尚未寄送或已超過 10 分鐘有效期限，請重新寄送驗證碼' };
  }

  const data = JSON.parse(raw);
  if (Date.now() > data.expiresAt) {
    cache.remove(key);
    return { ok: false, error: '驗證碼已過期，請重新寄送驗證碼' };
  }

  if (data.attempts >= 5) {
    cache.remove(key);
    return { ok: false, error: '驗證碼錯誤已達 5 次上限，請重新寄送新的驗證碼' };
  }

  if (inputCode !== data.code) {
    data.attempts += 1;
    const rem = Math.max(0, 5 - data.attempts);
    if (rem === 0) {
      cache.remove(key);
      return { ok: false, error: '驗證碼錯誤已達 5 次上限，請重新寄送新的驗證碼' };
    }
    cache.put(key, JSON.stringify(data), 600);
    return { ok: false, error: '6 位數驗證碼不正確，尚可嘗試 ' + rem + ' 次' };
  }

  cache.remove(key);
  return { ok: true, verified: true };
}

      : '';

    const htmlBody = [
      '<div style="font-family: \'Noto Sans TC\', \'Microsoft JhengHei\', sans-serif; max-width: 580px; margin: 0 auto; border: 1px solid #dfe4dc; border-radius: 10px; overflow: hidden; background: #fafaf5;">',
      '  <div style="background: #234c3c; color: #ffffff; padding: 18px 24px;">',
      '    <div style="font-size: 12px; opacity: 0.85;">康橋國際學校林口校區 × 舊鞋救命 Step30</div>',
      '    <div style="font-size: 18px; font-weight: bold; margin-top: 4px;">' + escapeHtmlGas(subject) + '</div>',
      '  </div>',
      '  <div style="padding: 24px; background: #ffffff; color: #243a32; line-height: 1.75; font-size: 14px;">',
      '    <p style="margin: 0 0 12px;"><strong>' + escapeHtmlGas(name) + '（' + escapeHtmlGas(rec.id || '') + '）</strong> 同學您好：</p>',
      '    <div style="background: #fafaf5; border: 1px solid #dfe4dc; border-radius: 8px; padding: 16px; white-space: pre-line;">' + escapeHtmlGas(message) + '</div>',
      '    ' + linkBlock,
      '  </div>',
      '  <div style="padding: 12px 24px; background: #f4f7f1; border-top: 1px solid #dfe4dc; font-size: 12px; color: #52625a;">',
      '    活動聯繫｜中學部學務處永續組 陳逸芸老師｜02-8512-8210',
      '  </div>',
      '</div>'
    ].join('\n');

    try {
      MailApp.sendEmail({
        to: email,
        subject: subject,
        body: name + ' 同學您好：\n\n' + message + (portalUrl ? '\n\n專區網址：' + portalUrl : ''),
        htmlBody: htmlBody,
        name: '康橋 × 舊鞋救命服務學習系統'
      });
      sentCount++;
      appendMailLog(rec.category || '任務通知', rec.id, name, email, subject, '已寄出');
    } catch (err) {
      failedList.push({ id: rec.id, name: name, reason: err.message || String(err) });
      appendMailLog(rec.category || '任務通知', rec.id, name, email, subject, '失敗：' + (err.message || err));
    }
  });

  return {
    ok: true,
    sentCount: sentCount,
    failedList: failedList,
    remainingQuota: MailApp.getRemainingDailyQuota()
  };
}

/**
 * 4. 試算表雙向同步：讀取與寫回
 */
function handleSyncPull() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName(MAIN_SHEET_NAME) || ss.getSheets()[0];
  const values = sheet.getDataRange().getValues();
  return {
    ok: true,
    sheetName: sheet.getName(),
    rows: values,
    syncedAt: Utilities.formatDate(new Date(), 'Asia/Taipei', 'yyyy/MM/dd HH:mm:ss')
  };
}

function handleSyncPush(payload) {
  const records = Array.isArray(payload.records) ? payload.records : [];
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName(MAIN_SHEET_NAME) || ss.getSheets()[0];
  const data = sheet.getDataRange().getValues();

  // 確保第 8~13 欄標題存在
  if (data.length > 0 && data[0].length < 13) {
    sheet.getRange(1, 8, 1, 6).setValues([['正取確認', '校內排班', '學習單分數', '11/21出席狀態', '核發總時數', '最後更新時間']]);
  }

  const idRowMap = {};
  for (let i = 1; i < data.length; i++) {
    const rowId = String(data[i][3] || '').trim().toUpperCase();
    if (rowId) idRowMap[rowId] = i + 1;
  }

  const nowStr = Utilities.formatDate(new Date(), 'Asia/Taipei', 'yyyy/MM/dd HH:mm');
  let updatedRows = 0;

  records.forEach(function(rec) {
    const key = String(rec.id || '').trim().toUpperCase();
    const targetRow = idRowMap[key];
    if (targetRow) {
      sheet.getRange(targetRow, 8, 1, 6).setValues([[
        rec.confirmed ? '已確認' : '未確認',
        rec.shiftsText || '',
        Number(rec.wsScore || 0),
        rec.attendanceLabel || '未點名',
        Number(rec.totalApprovedHours || 0),
        nowStr
      ]]);
      updatedRows++;
    }
  });

  return {
    ok: true,
    updatedRows: updatedRows,
    syncedAt: nowStr
  };
}

function appendMailLog(type, studentId, name, email, subject, status) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    let logSheet = ss.getSheetByName(LOG_SHEET_NAME);
    if (!logSheet) {
      logSheet = ss.insertSheet(LOG_SHEET_NAME);
      logSheet.appendRow(['寄送時間', '信件類型', '學號', '姓名', '收件學校信箱', '信件主旨', '發送狀態']);
    }
    const ts = Utilities.formatDate(new Date(), 'Asia/Taipei', 'yyyy/MM/dd HH:mm:ss');
    logSheet.appendRow([ts, type, studentId || '', name || '', email, subject, status]);
  } catch (e) {}
}

function escapeHtmlGas(str) {
  return String(str || '')
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;');
}
