const SPREADSHEET_ID = '1hOyDHewvzfxvnq3xioxO8KkBtogRZ2ypBI6wy1yabfk';
const MAIN_SHEET_NAME = '舊鞋救命錄取名單';
const STATE_SHEET_NAME = '系統雲端狀態庫';
const LOG_SHEET_NAME = '系統發信紀錄';
const CHUNK_SIZE = 40000;

function doGet(e) {
  const params = (e && e.parameter) ? e.parameter : {};
  if (!params.action) {
    return HtmlService.createHtmlOutputFromFile('index')
      .setTitle('康橋 × 舊鞋救命｜2026 服務學習')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
      .addMetaTag('viewport', 'width=device-width, initial-scale=1.0');
  }

  let result = { ok: false, error: '未知的請求動作' };
  try {
    if (params.action === 'ping') result = handlePing();
    else if (params.action === 'pullFullState') result = handlePullFullState();
    else if (params.action === 'syncPull') result = handleSyncPull();
  } catch (err) {
    result = { ok: false, error: err.message || String(err) };
  }

  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  let result = { ok: false, error: '無效的請求內容' };
  try {
    const raw = (e && e.postData && e.postData.contents) ? e.postData.contents : '{}';
    const payload = JSON.parse(raw);
    result = serverApiBridge(payload);
  } catch (err) {
    result = { ok: false, error: err.message || String(err) };
  }

  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function serverApiBridge(payload) {
  const action = (payload && payload.action) ? payload.action : '';
  if (action === 'ping') return handlePing();
  if (action === 'sendOtp') return handleSendOtp(payload);
  if (action === 'verifyOtp') return handleVerifyOtp(payload);
  if (action === 'sendNotice') return handleSendNotice(payload);
  if (action === 'pullFullState') return handlePullFullState();
  if (action === 'pushFullState') return handlePushFullState(payload);
  if (action === 'upsertStudent') return handleUpsertStudent(payload);
  if (action === 'syncPull') return handleSyncPull();
  if (action === 'syncPush') return handleSyncPush(payload);
  return { ok: false, error: '不支援的操作類型：' + action };
}

function readMasterStateFromSheet(ss) {
  const sheet = ss.getSheetByName(STATE_SHEET_NAME);
  if (!sheet) return null;
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return null;
  const values = sheet.getRange(2, 1, lastRow - 1, 2).getValues();
  let jsonStr = '';
  for (let i = 0; i < values.length; i++) {
    jsonStr += String(values[i][1] || '');
  }
  if (!jsonStr) return null;
  return JSON.parse(jsonStr);
}

function writeMasterStateToSheet(ss, stateObj) {
  let sheet = ss.getSheetByName(STATE_SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(STATE_SHEET_NAME);
  }
  const cleanState = JSON.parse(JSON.stringify(stateObj));
  const jsonStr = JSON.stringify(cleanState);
  const rows = [];
  for (let i = 0; i < jsonStr.length; i += CHUNK_SIZE) {
    rows.push([rows.length + 1, jsonStr.slice(i, i + CHUNK_SIZE)]);
  }
  sheet.clearContents();
  sheet.getRange(1, 1, 1, 3).setValues([['區塊序號', 'JSON資料庫內容（請勿手動修改）', '最後同步時間']]);
  if (rows.length > 0) {
    sheet.getRange(2, 1, rows.length, 2).setValues(rows);
    sheet.getRange(2, 3).setValue(Utilities.formatDate(new Date(), 'Asia/Taipei', 'yyyy/MM/dd HH:mm:ss'));
  }
  updateReadableRosterSheet(ss, cleanState.students || []);
}

function updateReadableRosterSheet(ss, students) {
  if (!Array.isArray(students) || students.length === 0) return;
  const sheet = ss.getSheetByName(MAIN_SHEET_NAME) || ss.getSheets()[0];
  const data = sheet.getDataRange().getValues();
  if (data.length > 0 && data[0].length < 13) {
    sheet.getRange(1, 8, 1, 6).setValues([['正取確認', '校內排班', '學習單分數', '11/21出席狀態', '核發總時數', '最後更新時間']]);
  }
  const idRowMap = {};
  for (let i = 1; i < data.length; i++) {
    const rowId = String(data[i][3] || '').trim().toUpperCase();
    if (rowId) idRowMap[rowId] = i + 1;
  }
  const attMap = { pending: '未點名', present: '有出席', late: '遲到', early: '早退', absent: '缺席' };
  const nowStr = Utilities.formatDate(new Date(), 'Asia/Taipei', 'yyyy/MM/dd HH:mm');

  students.forEach(function(s) {
    const key = String(s.id || '').trim().toUpperCase();
    const targetRow = idRowMap[key];
    if (targetRow) {
      const preApp = (s.wsLocked && Number(s.wsScore || 0) >= 80 && s.reviews && s.reviews.pre === 'passed') ? 1.0 : 0.0;
      const whApp = (s.attendance === 'present') ? 3.0 : (s.attendance === 'late' ? 2.5 : (s.attendance === 'early' ? Number(s.attHours || 2.0) : 0.0));
      const campApp = (s.shiftsLocked && s.reviews && s.reviews.campus === 'passed') ? (Array.isArray(s.shifts) ? s.shifts.length * 0.5 : 0) : 0.0;
      const refApp = (s.refLocked && s.reviews && s.reviews.reflection === 'passed') ? 1.0 : 0.0;
      const totalApp = preApp + whApp + campApp + refApp;

      sheet.getRange(targetRow, 8, 1, 6).setValues([[
        s.confirmed ? '已確認' : '未確認',
        Array.isArray(s.shifts) ? s.shifts.join('、') : '',
        Number(s.wsScore || 0),
        attMap[s.attendance] || '未點名',
        totalApp,
        nowStr
      ]]);
    }
  });
}

function handlePullFullState() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const state = readMasterStateFromSheet(ss);
  if (!state) {
    return { ok: true, empty: true, syncedAt: Utilities.formatDate(new Date(), 'Asia/Taipei', 'yyyy/MM/dd HH:mm:ss') };
  }
  return {
    ok: true,
    empty: false,
    state: state,
    syncedAt: Utilities.formatDate(new Date(), 'Asia/Taipei', 'yyyy/MM/dd HH:mm:ss')
  };
}

function handlePushFullState(payload) {
  if (!payload || !payload.state) return { ok: false, error: '缺少狀態資料' };
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    writeMasterStateToSheet(ss, payload.state);
    return {
      ok: true,
      syncedAt: Utilities.formatDate(new Date(), 'Asia/Taipei', 'yyyy/MM/dd HH:mm:ss')
    };
  } finally {
    lock.releaseLock();
  }
}

function handleUpsertStudent(payload) {
  const incoming = payload && payload.student;
  if (!incoming || !incoming.id) return { ok: false, error: '無效的學生資料' };

  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    let master = readMasterStateFromSheet(ss);
    if (!master && payload.fallbackState) {
      master = payload.fallbackState;
    }
    if (!master || !Array.isArray(master.students)) {
      return { ok: false, error: '雲端資料庫尚未初始化，請通知教師先同步名單' };
    }

    const idx = master.students.findIndex(function(s) {
      return String(s.id).toLowerCase() === String(incoming.id).toLowerCase();
    });
    if (idx >= 0) {
      master.students[idx] = incoming;
    } else {
      master.students.push(incoming);
    }

    writeMasterStateToSheet(ss, master);
    return {
      ok: true,
      state: master,
      syncedAt: Utilities.formatDate(new Date(), 'Asia/Taipei', 'yyyy/MM/dd HH:mm:ss')
    };
  } finally {
    lock.releaseLock();
  }
}

function normalizeStudentEmail(studentId, providedEmail) {
  const cleanEmail = String(providedEmail || '').trim().toLowerCase();
  if (cleanEmail && cleanEmail.indexOf('@') > 0) return cleanEmail;
  const cleanId = String(studentId || '').trim().toLowerCase();
  if (!cleanId) return '';
  if (cleanId === 'chunyichou' || cleanId === 'yiyunchen') return cleanId + '@kcislk.ntpc.edu.tw';
  return cleanId + '@stu.kcislk.ntpc.edu.tw';
}

function handleSendOtp(payload) {
  const studentId = String(payload.studentId || '').trim();
  const studentName = String(payload.studentName || '').trim() || '同學';
  const recipientEmail = normalizeStudentEmail(studentId, payload.email);
  if (!recipientEmail || !/@(stu\.)?kcislk\.ntpc\.edu\.tw$/i.test(recipientEmail)) {
    return { ok: false, error: '僅允許寄送至康橋學校信箱 (@stu.kcislk.ntpc.edu.tw)' };
  }
  const cache = CacheService.getScriptCache();
  if (cache.get('OTP_CD_' + recipientEmail)) {
    return { ok: false, error: '驗證碼寄送過於頻繁，請稍候 60 秒後再試' };
  }
  const code = String(Math.floor(100000 + Math.random() * 900000));
  cache.put('OTP_DATA_' + recipientEmail, JSON.stringify({ code: code, attempts: 0, expiresAt: Date.now() + 600000 }), 600);
  cache.put('OTP_CD_' + recipientEmail, '1', 60);

  const subject = '【康橋 × 舊鞋救命】學生專區登入驗證碼：' + code;
  MailApp.sendEmail({
    to: recipientEmail,
    subject: subject,
    body: studentName + ' 同學您好：\n\n您的登入驗證碼為：' + code + '（10 分鐘內有效）。',
    name: '康橋 × 舊鞋救命服務學習系統'
  });
  appendMailLog('OTP驗證碼', studentId, studentName, recipientEmail, subject, '已寄出');
  return { ok: true, sentTo: recipientEmail, remainingQuota: MailApp.getRemainingDailyQuota() };
}

function handleVerifyOtp(payload) {
  const recipientEmail = normalizeStudentEmail(payload.studentId, payload.email);
  const inputCode = String(payload.code || '').trim();
  const cache = CacheService.getScriptCache();
  const key = 'OTP_DATA_' + recipientEmail;
  const raw = cache.get(key);
  if (!raw) return { ok: false, error: '驗證碼尚未寄送或已過期，請重新寄送' };
  const data = JSON.parse(raw);
  if (Date.now() > data.expiresAt || data.attempts >= 5) {
    cache.remove(key);
    return { ok: false, error: '驗證碼已失效，請重新寄送' };
  }
  if (inputCode !== data.code) {
    data.attempts += 1;
    cache.put(key, JSON.stringify(data), 600);
    return { ok: false, error: '驗證碼不正確' };
  }
  cache.remove(key);
  return { ok: true, verified: true };
}

    const message = String(rec.message || '').trim();
    if (!email || !/@(stu\.)?kcislk\.ntpc\.edu\.tw$/i.test(email)) {
      failedList.push({ id: rec.id, name: name, reason: '信箱格式非康橋校網網域' });
      return;
    }
    try {
      MailApp.sendEmail({
        to: email,
        subject: subject,
        body: name + ' 同學您好：\n\n' + message + (portalUrl ? '\n\n專區網址：' + portalUrl : ''),
        name: '康橋 × 舊鞋救命服務學習系統'
      });
      sentCount++;
      appendMailLog(rec.category || '任務通知', rec.id, name, email, subject, '已寄出');
    } catch (err) {
      failedList.push({ id: rec.id, name: name, reason: err.message || String(err) });
    }
  });
  return { ok: true, sentCount: sentCount, failedList: failedList, remainingQuota: MailApp.getRemainingDailyQuota() };
}

function handleSyncPull() { return handlePullFullState(); }
function handleSyncPush(payload) { return handlePushFullState(payload); }

function appendMailLog(type, studentId, name, email, subject, status) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    let logSheet = ss.getSheetByName(LOG_SHEET_NAME);
    if (!logSheet) {
      logSheet = ss.insertSheet(LOG_SHEET_NAME);
      logSheet.appendRow(['寄送時間', '信件類型', '學號', '姓名', '收件學校信箱', '信件主旨', '發送狀態']);
    }
    logSheet.appendRow([Utilities.formatDate(new Date(), 'Asia/Taipei', 'yyyy/MM/dd HH:mm:ss'), type, studentId || '', name || '', email, subject, status]);
  } catch (e) {}
}
